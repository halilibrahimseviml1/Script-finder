-- Scriptleri Çekme Fonksiyonu
local function fetchGameScripts()
    local scriptsList = {}
    local seenIds = {}

    if gameName ~= "" and gameName ~= "Unknown Game" then
        local searchUrl = "https://scriptblox.com/api/script/search?q=" .. HttpService:UrlEncode(gameName) .. "&mode=free"
        local success, response = pcall(function()
            return game:HttpGet(searchUrl)
        end)
        
        if success and response then
            local data = HttpService:JSONDecode(response)
            if data and data.result and data.result.scripts then
                for _, s in ipairs(data.result.scripts) do
                    local sId = s._id or s.slug
                    if sId and not seenIds[sId] then
                        seenIds[sId] = true
                        table.insert(scriptsList, s)
                    end
                end
            end
        end
    end

    if #scriptsList == 0 then
        local fetchUrl = "https://scriptblox.com/api/script/fetch?placeId=" .. tostring(placeId)
        local success, response = pcall(function()
            return game:HttpGet(fetchUrl)
        end)
        
        if success and response then
            local data = HttpService:JSONDecode(response)
            if data and data.result and data.result.scripts then
                for _, s in ipairs(data.result.scripts) do
                    local sId = s._id or s.slug
                    if sId and not seenIds[sId] then
                        seenIds[sId] = true
                        table.insert(scriptsList, s)
                    end
                end
            end
        end
    end

    return scriptsList
end

-- Arayüz Listeleme Fonksiyonu
local showingFavoritesOnly = false

local function renderScripts(scriptsToRender, isFavTab)
    for _, child in ipairs(Scroll:GetChildren()) do
        if child:IsA("Frame") or child:IsA("TextLabel") then
            child:Destroy()
        end
    end

    if #scriptsToRender == 0 then
        local info = Instance.new("TextLabel")
        info.Size = UDim2.new(1, 0, 0, 30)
        info.BackgroundTransparency = 1
        info.Text = isFavTab and "⭐ No scripts have been added to favorites yet." or "❌ No script was found for this game."
        info.TextColor3 = Color3.fromRGB(130, 130, 150)
        info.TextSize = 11
        info.Parent = Scroll
        return
    end

    for _, item in ipairs(scriptsToRender) do
        local card = Instance.new("Frame")
        card.Size = UDim2.new(1, 0, 0, 44)
        card.BackgroundColor3 = Color3.fromRGB(18, 18, 26)
        card.Parent = Scroll

        local hasKey = item.isKey == true or item.key == true or (item.keyLink and item.keyLink ~= "")
        local displayTitle = item.title or "Untitled Script"
        
        if hasKey then
            displayTitle = displayTitle .. " KEY 🔑"
        end

        local sTitle = Instance.new("TextLabel")
        sTitle.Size = UDim2.new(1, -135, 1, 0)
        sTitle.Position = UDim2.new(0, 8, 0, 0)
        sTitle.BackgroundTransparency = 1
        sTitle.Text = displayTitle
        sTitle.TextColor3 = Color3.fromRGB(195, 195, 215)
        sTitle.TextSize = 10
        sTitle.Font = Enum.Font.SourceSans
        sTitle.TextTruncate = Enum.TextTruncate.AtEnd
        sTitle.TextXAlignment = Enum.TextXAlignment.Left
        sTitle.Parent = card

        local sId = tostring(item._id or item.slug or item.title)
        local isFav = favorites[sId] ~= nil

        local favBtn = Instance.new("TextButton")
        favBtn.Size = UDim2.new(0, 22, 0, 26)
        favBtn.Position = UDim2.new(1, -118, 0.5, -13)
        favBtn.BackgroundColor3 = isFav and Color3.fromRGB(120, 100, 20) or Color3.fromRGB(40, 40, 55)
        favBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        favBtn.Text = isFav and "⭐" or "☆"
        favBtn.TextSize = 10
        favBtn.Font = Enum.Font.SourceSansBold
        favBtn.Parent = card

        local runBtn = Instance.new("TextButton")
        runBtn.Size = UDim2.new(0, 42, 0, 26)
        runBtn.Position = UDim2.new(1, -93, 0.5, -13)
        runBtn.BackgroundColor3 = Color3.fromRGB(30, 110, 60)
        runBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        runBtn.Text = "▶ Run"
        runBtn.TextSize = 9
        runBtn.Font = Enum.Font.SourceSansBold
        runBtn.Parent = card

        local copyBtn = Instance.new("TextButton")
        copyBtn.Size = UDim2.new(0, 44, 0, 26)
        copyBtn.Position = UDim2.new(1, -46, 0.5, -13)
        copyBtn.BackgroundColor3 = Color3.fromRGB(35, 75, 140)
        copyBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        copyBtn.Text = "📋 Copy"
        copyBtn.TextSize = 9
        copyBtn.Font = Enum.Font.SourceSansBold
        copyBtn.Parent = card

        local scriptCode = item.script or item.code or ""

        favBtn.MouseButton1Click:Connect(function()
            if favorites[sId] then
                favorites[sId] = nil
                favBtn.Text = "☆"
                favBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 55)
            else
                favorites[sId] = item
                favBtn.Text = "⭐"
                favBtn.BackgroundColor3 = Color3.fromRGB(120, 100, 20)
            end
            saveFavorites()
            
            if showingFavoritesOnly then
                local favList = {}
                for _, favItem in pairs(favorites) do
                    table.insert(favList, favItem)
                end
                renderScripts(favList, true)
            end
        end)

        runBtn.MouseButton1Click:Connect(function()
            if scriptCode ~= "" then
                local success, err = pcall(function()
                    loadstring(scriptCode)()
                end)
                if not success then
                    runBtn.Text = "⚠️ ErrorLight"
                    task.wait(1)
                    runBtn.Text = "▶ Run"
                end
            end
        end)

        copyBtn.MouseButton1Click:Connect(function()
            if scriptCode ~= "" then -- DÜZELTİLDİ: scriptCode
                if setclipboard then
                    setclipboard(scriptCode)
                    copyBtn.Text = "✅"
                    task.wait(1)
                    copyBtn.Text = "📋 Copy"
                else
                    copyBtn.Text = "❌"
                    task.wait(1)
                    copyBtn.Text = "📋 Copy"
                end
            end
        end)
    end
end
